**In [99]:**

```python
# question wants the order too
remaining_features = list(X.columns)
chosen_features = []
selection_order = []

best_score = 0

while remaining_features:

    feature_scores = {}

    for feature in remaining_features:

        current_features = chosen_features + [feature]

        model = LogisticRegression(
            max_iter=1000
        )

        model.fit(
            X_train_scaled[current_features],
            y_train
        )

        predictions = model.predict(
            X_test_scaled[current_features]
        )

        score = accuracy_score(
            y_test,
            predictions
        )

        feature_scores[feature] = score

    best_feature = max(
        feature_scores,
        key=feature_scores.get
    )

    current_score = feature_scores[
        best_feature
    ]

    # Stop if adding features no longer improves accuracy
    if current_score < best_score:
        break

    chosen_features.append(best_feature)
    remaining_features.remove(best_feature)

    selection_order.append(
        (best_feature, current_score)
    )

    best_score = current_score
```

**In [164]:**

```python
# How is SMOTE different from Random Oversampling?
# Random Oversampling balances the classes by randomly duplicating existing minority-class 
# observations. SMOTE instead creates new synthetic minority-class samples by interpolating 
# between neighbouring minority observations.
```

**In [166]:**

```python
# Why is SMOTE generally preferred over simply duplicating samples?
# SMOTE is generally preferred because it introduces new synthetic samples rather 
# than repeatedly copying the same observations. This provides greater diversity in 
# the minority class and can reduce the risk of the model memorizing duplicated 
# observations. However, SMOTE should still be used carefully because synthetic 
# observations may sometimes introduce noise.
```
